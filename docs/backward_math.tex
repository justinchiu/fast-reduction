\section{Notation}
Let
\begin{align}
X &\in \mathbb{R}^{B \times d}, &
W &\in \mathbb{R}^{V \times d}, &
b &\in \mathbb{R}^{V}, \\
Z &= XW^\top + \mathbf{1}b^\top \in \mathbb{R}^{B \times V}.
\end{align}
For row $n$ and class $j$:
\begin{align}
p_{nj} &= \mathrm{softmax}(z_n)_j
= \frac{e^{z_{nj}}}{\sum_k e^{z_{nk}}}, \\
\log p_{nj} &= z_{nj} - \log \sum_k e^{z_{nk}}.
\end{align}

\section{Fast-Reduction: CE + Entropy + Log-Prob Backward}
\subsection{Math}
Per-row outputs:
\begin{align}
\ell_n^{\mathrm{ce}} &= -\log p_{n,t_n}, \\
\ell_n^{\mathrm{ent}} &= -\sum_j p_{nj}\log p_{nj}, \\
\ell_n^{\log p} &= \log p_{n,t_n} = -\ell_n^{\mathrm{ce}}.
\end{align}

Given upstream row gradients
$g_n^{\mathrm{ce}}, g_n^{\mathrm{ent}}, g_n^{\log p}$, define
\begin{align}
\tilde g_n^{\mathrm{ce}} = g_n^{\mathrm{ce}} - g_n^{\log p}.
\end{align}

Then
\begin{align}
\frac{\partial L}{\partial z_{nj}}
=
\tilde g_n^{\mathrm{ce}}
\left(p_{nj} - \mathbf{1}[j=t_n]\right)
+ g_n^{\mathrm{ent}}
\left(-p_{nj}\left(\log p_{nj} + H_n\right)\right),
\end{align}
where
\begin{align}
H_n = -\sum_j p_{nj}\log p_{nj}.
\end{align}

Matrix form:
\begin{align}
D_Z[n,j]
=
\tilde g_n^{\mathrm{ce}}\left(p_{nj} - Y_{nj}\right)
+ g_n^{\mathrm{ent}}\left(-p_{nj}(\log p_{nj} + H_n)\right).
\end{align}

If a row is ignored (e.g.\ $t_n < 0$), the CE/log-prob term is zeroed.

Linear gradients:
\begin{align}
D_X &= D_Z W, \\
D_W &= D_Z^\top X, \\
D_b &= \sum_{n=1}^B D_Z[n,:].
\end{align}

\subsection{How It Is Realized In Code}
In \texttt{fast\_reduction/kernel.py}, function
\texttt{\_compute\_dlogits\_chunk}:
\begin{itemize}
\item \texttt{lse = logsumexp(logits\_fp32)} and
\texttt{logits\_fp32.sub\_(lse)} compute $\log p$ in-place.
\item \texttt{row\_a = g\_ce - g\_ent * entropy}, \texttt{row\_b = -g\_ent}
construct per-row coefficients so that
\begin{align}
D_Z = \exp(\log P)\odot(\texttt{row\_a} + \texttt{row\_b}\odot\log P).
\end{align}
\item \texttt{dlogits[arange, target] -= g\_ce} applies the
$-\tilde g_n^{\mathrm{ce}}\mathbf{1}[j=t_n]$ term.
\end{itemize}

Then \texttt{fused\_linear\_xent\_entropy\_backward} performs chunked GEMMs:
\begin{align}
D_X^{(c)} &= D_Z^{(c)} W, \\
D_W &\mathrel{+}= (D_Z^{(c)})^\top X^{(c)}, \\
D_b &\mathrel{+}= \sum_{n\in c} D_Z[n,:].
\end{align}

\section{Quack: Chunked Linear Cross-Entropy Backward}
\subsection{Math}
Objective (no entropy term):
\begin{align}
\ell_n = -\log p_{n,t_n}, \qquad
L_{\mathrm{sum}} = \sum_{n\in\mathcal{V}} \ell_n,
\end{align}
with $\mathcal{V}$ excluding \texttt{ignore\_index}.
For CE:
\begin{align}
\frac{\partial \ell_n}{\partial z_{nj}} = p_{nj} - \mathbf{1}[j=t_n].
\end{align}

\subsection{How It Is Realized In Code}
In \texttt{quack/linear\_cross\_entropy.py},
\texttt{chunked\_linear\_cross\_entropy\_fwd} does, per chunk $c$:
\begin{align}
Z_c &= X_c W^\top, \\
D_c &= \frac{\partial \ell_c}{\partial Z_c}
\quad\text{(written in-place into logits buffer)}, \\
\widehat D_{X,c} &= D_c W.
\end{align}
This uses \texttt{cross\_entropy\_fwd\_out(..., dx=dlogits\_chunk)} so CE
gradient is produced during forward.

Weight gradient accumulation is split:
\begin{align}
G_{\mathrm{partial}} &= \sum_{c\neq c_{\mathrm{last}}} D_c^\top X_c, \\
\text{save }(D_{c_{\mathrm{last}}}, X_{c_{\mathrm{last}}})&\text{ for backward}.
\end{align}

In \texttt{ChunkedLinearCrossEntropyFunction.backward}, if scalar upstream is
$g$, with reduction scale $s$ (\texttt{sum}: $1$, \texttt{mean}: $1/|\mathcal{V}|$):
\begin{align}
\alpha &= gs, \\
D_X &= \alpha\,\widehat D_X, \\
D_W &= \alpha\left(G_{\mathrm{partial}} + D_{c_{\mathrm{last}}}^\top X_{c_{\mathrm{last}}}\right).
\end{align}
So backward is mostly scaling and finishing deferred $D_W$.

For CE kernel internals, \texttt{quack/cross\_entropy.py}
(\texttt{CrossEntropyBackward.kernel}) computes
\begin{align}
P = \exp(z-\mathrm{lse}),\quad
D_Z = \mathrm{where}(j=t,\;P-1,\;P)\cdot d\ell,
\end{align}
via a vectorized boolean mask and \texttt{cute.where}.

\section{Liger-Kernel: Fused Linear Cross-Entropy Backward}
\subsection{Math}
Core CE piece is the same:
\begin{align}
\frac{\partial \ell_n}{\partial z_{nj}} = p_{nj} - \mathbf{1}[j=t_n],
\end{align}
with optional additions for class weights, label smoothing, softcapping, and
z-loss.

\subsection{How It Is Realized In Code}
In \texttt{liger\_kernel/ops/fused\_linear\_cross\_entropy.py},
\texttt{fused\_linear\_cross\_entropy\_forward} loops over token chunks and:
\begin{align}
Z_c &= X_c W^\top (+b), \\
D_c &= \frac{\partial \ell_c}{\partial Z_c}
\quad\text{(written in-place by Triton kernel)}, \\
D_{X,c} &= D_c W, \\
D_W &\mathrel{+}= D_c^\top X_c, \\
D_b &\mathrel{+}= \sum_{n\in c} D_c[n,:].
\end{align}

The Triton kernel \texttt{liger\_cross\_entropy\_kernel} in
\texttt{liger\_kernel/ops/cross\_entropy.py} uses:
\begin{itemize}
\item Pass 1: online softmax stats $(m,d)$ and optional argmax.
\item Pass 2: writes $\partial L/\partial Z$ back to \texttt{X\_ptr}.
\item Then computes/stores loss (plus optional z-loss/metrics).
\end{itemize}

Autograd behavior:
\begin{itemize}
\item \texttt{LigerFusedLinearCrossEntropyFunction.forward} stores
precomputed $(D_X,D_W,D_b)$.
\item \texttt{backward} calls
\texttt{fused\_linear\_cross\_entropy\_backward}, which only rescales stored
grads by scalar upstream \texttt{grad\_output} (using
\texttt{element\_mul\_kernel}); if \texttt{grad\_output == 1}, scaling is skipped.
\end{itemize}

\section{Quack Plain Linear Backward (Reference)}
For plain linear $Y=XW^\top+b$, quack's \texttt{linear.py} follows:
\begin{align}
D_X &= D_Y W, \\
D_W &= D_Y^\top X, \\
D_b &= \sum_n D_Y[n,:],
\end{align}
with an optional in-place accumulation path for \texttt{weight.grad}; math is unchanged.
